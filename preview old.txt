    <script>
        async function sortColors() {
            const n = document.getElementById('nInput').value;
            const colors = document.getElementById('colorsInput').value;
            const resultContainer = document.getElementById('resultContainer');
            const outputMessage = document.getElementById('outputMessage');
            const visualResult = document.getElementById('visualResult');

            // Reset UI
            outputMessage.className = "font-mono text-lg break-words text-slate-300";
            visualResult.innerHTML = '';
            
            if (!n || !colors) {
                resultContainer.classList.remove('hidden');
                outputMessage.textContent = "Please fill in all fields";
                outputMessage.classList.add('text-red-400');
                return;
            }

            try {
                const response = await fetch('/api/sort', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ n, colors })
                });

                const data = await response.json();

                resultContainer.classList.remove('hidden');
                outputMessage.textContent = data.result;

                if (data.result === 'sorry') {
                    outputMessage.classList.add('text-red-500', 'font-bold');
                } else if (data.success) {
                    // Create visual blocks if successful
                    const nums = data.result.split(' ');
                    nums.forEach(num => {
                        const div = document.createElement('div');
                        div.className = `w-8 h-8 rounded-md flex items-center justify-center font-bold text-sm shadow-md transition-transform hover:scale-110 ${getColorClass(num)}`;
                        div.textContent = num;
                        visualResult.appendChild(div);
                    });
                }

            } catch (error) {
                console.error('Error:', error);
                resultContainer.classList.remove('hidden');
                outputMessage.textContent = "Error connecting to server.";
                outputMessage.classList.add('text-red-400');
            }
        }

        function getColorClass(num) {
            if (num === '0') return 'bg-red-500 text-white border border-red-400';
            if (num === '1') return 'bg-white text-slate-900 border border-slate-300';
            if (num === '2') return 'bg-blue-500 text-white border border-blue-400';
            return 'bg-slate-700 text-slate-400';
        }
    </script>
